shader_type canvas_item;

uniform vec4 glow_color : source_color = vec4(0.4, 0.6, 1.0, 0.5);
uniform float glow_size : hint_range(0.0, 20.0) = 5.0;

void fragment() {
	vec4 color = texture(TEXTURE, UV);
	if (color.a > 0.1) {
		COLOR = color + glow_color * glow_size * color.a;
	} else {
		COLOR = color;
	}
}
